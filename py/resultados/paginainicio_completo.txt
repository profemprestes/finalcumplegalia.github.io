<script>
  document.addEventListener('DOMContentLoaded', () => {
    const EVENT_DATE: Date = new Date('2025-05-10T13:00:00-03:00');
    const elements: Record<string, HTMLElement | null> = {
        days: document.getElementById('countdown-days'),
        hours: document.getElementById('countdown-hours'),
        minutes: document.getElementById('countdown-minutes'),
        seconds: document.getElementById('countdown-seconds')
    };
    const container: HTMLElement | null = document.querySelector('.countdown-container');
    const messageElement: HTMLElement | null = document.getElementById('countdown-message');

    interface TimeRemaining {
        expired: boolean;
        days?: number;
        hours?: number;
        minutes?: number;
        seconds?: number;
    }

    function getTimeRemaining(): TimeRemaining {
        const now: Date = new Date();
        const diff: number = EVENT_DATE.getTime() - now.getTime();

        if (diff < 0) return { expired: true };
        return {
            expired: false,
            days: Math.floor(diff / (1000 * 60 * 60 * 24)),
            hours: Math.floor((diff / (1000 * 60 * 60)) % 24),
            minutes: Math.floor((diff / (1000 * 60)) % 60),
            seconds: Math.floor((diff / 1000) % 60)
        };
    }

    function updateCountdown(): void {
        const time: TimeRemaining = getTimeRemaining();
        if (time.expired && container && messageElement) {
            container.style.display = 'none';
            messageElement.style.display = 'block';
            return;
        }

        Object.entries(elements).forEach(([key, el]) => {
            if (el && time[key as keyof TimeRemaining]) {
                el.textContent = String(time[key as keyof TimeRemaining]).padStart(2, '0');
            }
        });
    }

    updateCountdown();
    const interval = setInterval(updateCountdown, 1000);

    window.addEventListener('beforeunload', () => clearInterval(interval));
  });

  function scrollTo(sectionId: string): void {
    const section = document.getElementById(sectionId);
    if (section) section.scrollIntoView({ behavior: 'smooth' });
  }
</script>
